vim.api.nvim_create_autocmd("UIEnter", {
  once = true,
  callback = function()
    require("vim._core.ui2").enable({
      enable = true,
      msg = {
        targets = {
          [''] = 'msg', empty = 'cmd', bufwrite = 'msg', confirm = 'cmd',
          emsg = 'pager', echo = 'msg', echomsg = 'msg', echoerr = 'pager',
        }
      }
    })
  end,
})

-- Custom minimalist color palette
local function apply_strict_colors()
  local c = { 
    bg     = "#080808", fg     = "#d0d0d0", gray   = "#6b6b6b",
    blue   = "#768ec7", green  = "#81b285", yellow = "#d3ab68", red    = "#c46c6c"
  }
  local hl = vim.api.nvim_set_hl
  
  hl(0, "Normal", { fg = c.fg, bg = c.bg })
  hl(0, "LineNr", { fg = c.gray })
  hl(0, "SignColumn", { bg = c.bg })
  hl(0, "Comment", { fg = c.gray, italic = true })
  hl(0, "String", { fg = c.green })
  hl(0, "Function", { fg = c.yellow })
  hl(0, "Statement", { fg = c.blue })
  hl(0, "Keyword", { fg = c.blue })
  hl(0, "Type", { fg = c.blue })
  hl(0, "PreProc", { fg = c.blue })
  
  hl(0, "StatusLine", { fg = c.gray, bg = c.bg, bold = true })
  hl(0, "StatusLineNC", { fg = c.gray, bg = c.bg })
  hl(0, "WinSeparator", { fg = c.gray, bg = c.bg })
end
apply_strict_colors()

-- Pressing <leader>h will instantly toggle inlay hints on and off
vim.keymap.set("n", "<leader>h", function()
    local current_setting = vim.lsp.inlay_hint.is_enabled({ bufnr = 0 })
    vim.lsp.inlay_hint.enable(not current_setting, { bufnr = 0 })
end, { desc = "Toggle Inlay Hints" })

-- LSP and code navigation
vim.api.nvim_create_autocmd('LspAttach', {
  callback = function(args)
    local client = vim.lsp.get_client_by_id(args.data.client_id)
    local bufnr = args.buf

    -- Protect the custom palette from semantic token overrides
    if client then client.server_capabilities.semanticTokensProvider = nil end
    
    -- Enable native autocompletion (<C-x><C-o>)
    vim.bo[bufnr].omnifunc = 'v:lua.vim.lsp.omnifunc'

    -- 'gd', 'gD', 'gr', 'gi', and 'K' are natively mapped in Neovim 0.10+
    local opts = { buffer = bufnr, silent = true }
    vim.keymap.set('n', '<leader>rn', vim.lsp.buf.rename, opts)
    vim.keymap.set('n', '<leader>ca', vim.lsp.buf.code_action, opts)
  end,
})

  }
})
